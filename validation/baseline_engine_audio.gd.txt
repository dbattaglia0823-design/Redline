class_name EngineAudio
extends AudioStreamPlayer

var playback: AudioStreamGeneratorPlayback
var phase := 0.0
var turbo_phase := 0.0
var smooth_rpm := 950.0
var pop := 0.0
var enabled := true
var noise_state := 0.0
const RATE := 22050.0

func _ready() -> void:
	var generator := AudioStreamGenerator.new()
	generator.mix_rate = RATE
	generator.buffer_length = 0.12
	stream = generator
	volume_db = -13.0
	play()
	playback = get_stream_playback()

func shift_pop() -> void:
	pop = 0.85

func fill(rpm: float, load_value: float, boost: float, cutting: bool, slip: float, nitrous: bool = false) -> void:
	if playback == null:
		return
	var count := mini(playback.get_frames_available(), 4096)
	for i in count:
		smooth_rpm += (rpm - smooth_rpm) * 0.0025
		# Four-cylinder four-stroke firing order: two combustion pulses per revolution.
		phase = fmod(phase + smooth_rpm / 60.0 * 2.0 / RATE, 1.0)
		var p := phase * TAU
		var rumble := sin(p) * 0.53 + sin(p * 2.0 + 0.3) * 0.24 + sin(p * 3.0) * 0.13 + sin(p * 5.0) * 0.07
		var noise := randf_range(-1.0, 1.0)
		noise_state = lerpf(noise_state, noise, 0.32)
		var body := tanh(rumble * (1.3 + load_value * 1.8)) * (0.28 + load_value * 0.52)
		if cutting:
			body *= 0.30
		turbo_phase = fmod(turbo_phase + (1100.0 + boost * 2400.0) / RATE, 1.0)
		var whistle := sin(turbo_phase * TAU) * boost * 0.055
		var exhaust := noise_state * (0.025 + load_value * 0.09 + slip * 0.1 + (0.11 if nitrous else 0.0))
		var crackle := noise * pop * 0.6
		pop *= 0.9990
		var sample := clampf(body + exhaust + whistle + crackle, -0.95, 0.95) if enabled else 0.0
		playback.push_frame(Vector2(sample, sample * 0.96))
